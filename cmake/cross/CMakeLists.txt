# Cross-compile check: compiles libobs for another OS from this host, with the
# LLVM and SDK soldr provisions (cmake/common/soldr.cmake) and the same
# prebuilt dependencies that OS's real build uses (CMakePresets.json, .deps/).
# It compiles every libobs source that OS builds, both the C build's ported
# files and the Rust build's C halves, and links nothing.
#
#   cmake -S cmake/cross -B build_cross_win-x64 -G Ninja -DOBS_CROSS_TARGET=win-x64
#   cmake --build build_cross_win-x64
#
# or, from a configured OBS build: cmake --build <build> --target cross-win-x64
# (cross-all for every alias). Aliases: win-x64, mac-arm64, mac-x64.
# Warnings are reported; -DOBS_CROSS_WERROR=ON makes them errors.

cmake_minimum_required(VERSION 3.28...3.30)

get_filename_component(OBS_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${OBS_SOURCE_ROOT}/cmake/common")
include(soldr)

set(OBS_CROSS_TARGET "" CACHE STRING "Cross target alias: win-x64, mac-arm64 or mac-x64")
# Also from the environment: when the toolchain changes the compiler, CMake
# resets the cache and reconfigures without the -D options it was given.
if(NOT OBS_CROSS_TARGET AND DEFINED ENV{OBS_CROSS_TARGET})
  set(OBS_CROSS_TARGET "$ENV{OBS_CROSS_TARGET}" CACHE STRING "Cross target alias" FORCE)
endif()
if(NOT OBS_CROSS_TARGET)
  string(REPLACE ";" ", " _aliases "${OBS_CROSS_ALIASES}")
  message(FATAL_ERROR "Set -DOBS_CROSS_TARGET=<alias> (${_aliases})")
endif()
option(OBS_CROSS_WERROR "Treat warnings as errors in the cross-compile check" OFF)

# The generated toolchain is rewritten when soldr.cmake (how it is made) or the
# alias changes, so a fix there reaches an existing build tree. A toolchain
# file passed in by the user is left alone.
set(_toolchain "${CMAKE_BINARY_DIR}/obs-cross-toolchain.cmake")
if(NOT CMAKE_TOOLCHAIN_FILE OR CMAKE_TOOLCHAIN_FILE STREQUAL _toolchain)
  file(SHA256 "${OBS_SOURCE_ROOT}/cmake/common/soldr.cmake" _toolchain_key)
  string(APPEND _toolchain_key "-${OBS_CROSS_TARGET}")
  if(NOT EXISTS "${_toolchain}" OR NOT OBS_CROSS_TOOLCHAIN_KEY STREQUAL _toolchain_key)
    obs_soldr_cross_toolchain(${OBS_CROSS_TARGET} "${_toolchain}")
    set(
      OBS_CROSS_TOOLCHAIN_KEY
      "${_toolchain_key}"
      CACHE INTERNAL
      "soldr.cmake hash and alias of the generated toolchain"
    )
  endif()
  set(CMAKE_TOOLCHAIN_FILE "${_toolchain}" CACHE FILEPATH "Generated by cmake/common/soldr.cmake" FORCE)
endif()

if(OBS_CROSS_TARGET MATCHES "^mac-")
  project(obs-cross LANGUAGES C CXX OBJC OBJCXX)
  set(_host MACOS)
  set(_deps_platform macos-universal)
  set(_deps_arch universal)
  set(_deps_file "macos-deps-VERSION-ARCH-REVISION.tar.xz")
else()
  project(obs-cross LANGUAGES C CXX)
  set(_host WINDOWS)
  string(REGEX REPLACE "^win-" "" _deps_arch "${OBS_CROSS_TARGET}")
  set(_deps_platform windows-${_deps_arch})
  set(_deps_file "windows-deps-VERSION-ARCH-REVISION.zip")
endif()

# The prebuilt dependencies the real build uses (buildspec_common.cmake reads
# the pins and hashes from CMakePresets.json under CMAKE_CURRENT_SOURCE_DIR).
include(buildspec_common)
function(_obs_cross_dependencies)
  set(CMAKE_CURRENT_SOURCE_DIR "${OBS_SOURCE_ROOT}")
  set(dependencies_dir "${OBS_SOURCE_ROOT}/.deps")
  set(arch ${_deps_arch})
  set(platform ${_deps_platform})
  set(prebuilt_filename "${_deps_file}")
  set(prebuilt_destination "obs-deps-VERSION-ARCH")
  _check_dependencies(prebuilt)
  _get_dependency_data(data)
  string(JSON data GET ${data} prebuilt)
  string(JSON version GET ${data} version)
  set(OBS_CROSS_DEPS "${dependencies_dir}/obs-deps-${version}-${arch}" PARENT_SCOPE)
endfunction()
_obs_cross_dependencies()
if(NOT EXISTS "${OBS_CROSS_DEPS}/include")
  message(FATAL_ERROR "Prebuilt dependencies for ${OBS_CROSS_TARGET} missing: ${OBS_CROSS_DEPS}/include")
endif()

function(_obs_cross_object_library name)
  set(files ${ARGN})
  list(FILTER files INCLUDE REGEX "\\.(c|m|cpp)$")
  list(TRANSFORM files PREPEND "${OBS_SOURCE_ROOT}/libobs/")
  add_library(${name} OBJECT ${files})
  target_include_directories(
    ${name}
    PRIVATE
      "${OBS_SOURCE_ROOT}/libobs"
      "${CMAKE_BINARY_DIR}/config"
      "${OBS_SOURCE_ROOT}/deps/libcaption"
      "${OBS_CROSS_DEPS}/include"
      $<$<STREQUAL:${_host},WINDOWS>:${OBS_SOURCE_ROOT}/deps/w32-pthreads>
  )
  target_compile_definitions(
    ${name}
    PRIVATE
      IS_LIBOBS
      ENABLE_HEVC
      $<$<STREQUAL:${_host},WINDOWS>:NOMINMAX;UNICODE;_UNICODE;_CRT_SECURE_NO_WARNINGS;_CRT_NONSTDC_NO_WARNINGS>
  )
  target_compile_options(${name} PRIVATE "$<$<COMPILE_LANGUAGE:C,OBJC>:${_obs_clang_c_options}>")
  set_target_properties(
    ${name}
    PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED TRUE COMPILE_WARNING_AS_ERROR ${OBS_CROSS_WERROR}
  )
endfunction()

# The platform-only plugins, compile-only like libobs above: each one's own
# sources with the libobs, shared/ and deps/ headers it builds against, through
# the same precompiled headers and unity batches as the native build. This is
# where Windows- and macOS-only file-local name collisions show up.
function(_obs_cross_plugin name)
  cmake_parse_arguments(PARSE_ARGV 1 _P "OBJC_ARC;OBJC_MODULES" "DIR" "SOURCES;INCLUDES;DEFINES")
  list(TRANSFORM _P_SOURCES PREPEND "${OBS_SOURCE_ROOT}/${_P_DIR}/")
  list(TRANSFORM _P_INCLUDES PREPEND "${OBS_SOURCE_ROOT}/")
  add_library(${name} OBJECT ${_P_SOURCES})
  target_include_directories(
    ${name}
    PRIVATE
      "${OBS_SOURCE_ROOT}/${_P_DIR}"
      "${CMAKE_BINARY_DIR}/${name}"
      ${_P_INCLUDES}
      "${OBS_SOURCE_ROOT}/libobs"
      "${CMAKE_BINARY_DIR}/config"
      "${OBS_CROSS_DEPS}/include"
      $<$<STREQUAL:${_host},WINDOWS>:${OBS_SOURCE_ROOT}/deps/w32-pthreads>
  )
  target_compile_definitions(
    ${name}
    PRIVATE
      ENABLE_HEVC
      OBS_VERSION="cross-check"
      ${_P_DEFINES}
      $<$<STREQUAL:${_host},WINDOWS>:NOMINMAX;UNICODE;_UNICODE;_CRT_SECURE_NO_WARNINGS;_CRT_NONSTDC_NO_WARNINGS>
  )
  target_compile_options(
    ${name}
    PRIVATE
      "$<$<COMPILE_LANGUAGE:C,OBJC>:${_obs_clang_c_options}>"
      "$<$<COMPILE_LANGUAGE:CXX,OBJCXX>:${_obs_clang_cxx_options}>"
  )
  # The Xcode settings the plugin's CMakeLists.txt sets
  # (XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC / _MODULES), as compiler flags.
  if(_P_OBJC_ARC)
    target_compile_options(${name} PRIVATE "$<$<COMPILE_LANGUAGE:OBJC,OBJCXX>:-fobjc-arc>")
  endif()
  if(_P_OBJC_MODULES)
    target_compile_options(${name} PRIVATE "$<$<COMPILE_LANGUAGE:OBJC,OBJCXX>:-fmodules>")
  endif()
  set_target_properties(
    ${name}
    PROPERTIES
      C_STANDARD 17
      C_STANDARD_REQUIRED TRUE
      CXX_STANDARD 17
      CXX_STANDARD_REQUIRED TRUE
      COMPILE_WARNING_AS_ERROR ${OBS_CROSS_WERROR}
      OBS_PCH_PROFILE plugin
  )
endfunction()
